# Staging stack — SMS / OTP testing.
#
# Brings up Postgres 16 + Redis 7 + the NestJS backend (behind Caddy) with the
# schema already migrated, so the stack is ready to send real Kavenegar SMS.
#
#   1. Put the staging secrets in a repo-root .env (it is gitignored, same as
#      backend/.env):
#
#        KAVENEGAR_API_KEY=<real key>
#        OTP_PEPPER=<random string>
#        JWT_ACCESS_SECRET=<random string>
#        JWT_REFRESH_SECRET=<random string>
#
#   2. docker compose -f docker-compose.staging.yml up -d --build
#
# Host ports are offset from the dev docker-compose.yml (5434/6381/3001) so
# staging can run next to the local dev stack without clashing.

services:
  postgres:
    image: postgres:16-alpine
    container_name: boya_postgres_staging
    restart: unless-stopped
    # 4GB single-host budget: shared_buffers + effective_cache_size reserve ~1.5GB
    # of the actual data; the container itself is capped at 1GB.
    # Same tuning as dev/production: without these flags Postgres falls back to
    # the 128MB shared_buffers default, which is too small for the RLS index
    # plans the panel relies on.
    command: ["postgres", "-c", "shared_buffers=1GB", "-c", "effective_cache_size=2GB", "-c", "work_mem=16MB", "-c", "maintenance_work_mem=128MB", "-c", "max_connections=50", "-c", "random_page_cost=1.1"]
    mem_limit: 1024m
    cpus: 1.5
    environment:
      POSTGRES_USER: beauty
      POSTGRES_PASSWORD: change-me
      POSTGRES_DB: beauty_platform
    ports:
      - "127.0.0.1:5434:5432" # localhost only; no external access to the data plane.
    volumes:
      - postgres_staging_data:/var/lib/postgresql/data
      - ./docker/postgres/init:/docker-entrypoint-initdb.d:ro
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U beauty -d beauty_platform"]
      interval: 5s
      timeout: 5s
      retries: 10

  redis:
    image: redis:7-alpine
    container_name: boya_redis_staging
    restart: unless-stopped
    mem_limit: 256m
    command: redis-server --maxmemory 256mb --maxmemory-policy allkeys-lru
    ports:
      - "127.0.0.1:6381:6379" # localhost only; no external access to the data plane.
    volumes:
      - redis_staging_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 10

  # One-shot: applies the TypeORM migration chain to the staging database. It
  # runs in the builder stage because `npm run migration:run` needs the TS
  # sources + ts-node, which the slim runtime image deliberately does not ship.
  # data-source.migration.ts refuses to run as beauty_app, so the master user
  # (beauty) is used here.
  backend-migrate:
    build:
      context: ./backend
      target: builder
    container_name: boya_backend_migrate_staging
    restart: "no"
    depends_on:
      postgres:
        condition: service_healthy
    environment:
      DATABASE_MIGRATION_HOST: postgres
      DATABASE_MIGRATION_PORT: "5432"
      DATABASE_MIGRATION_USER: beauty
      DATABASE_MIGRATION_PASSWORD: change-me
      DATABASE_MIGRATION_NAME: beauty_platform
    command: ["npm", "run", "migration:run"]

  backend:
    build:
      context: ./backend
    container_name: boya_backend_staging
    restart: unless-stopped
    depends_on:
      postgres:
        condition: service_healthy

        condition: service_healthy

        condition: service_completed_successfully

      # --- PostgreSQL (compose-internal network) ---
      DATABASE_HOST: postgres

      DATABASE_USER: beauty
      DATABASE_PASSWORD: change-me
      DATABASE_NAME: beauty_platform
      DATABASE_MIGRATION_USER: beauty
      DATABASE_MIGRATION_PASSWORD: change-me
      DATABASE_SSL: "false"
      # --- Redis ---
      REDIS_HOST: redis
      REDIS_PORT: "6379"

    }
